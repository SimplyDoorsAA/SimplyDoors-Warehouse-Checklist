function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = e.parameter; const timestamp = new Date(); let sheetName = ""; let headers = []; let rowData = []; let emailSubject = ""; let emailBody = ""; // Process up to 3 Images let photoUrls = ["N/A", "N/A", "N/A"]; const photoKeys = [data.PhotoData1, data.PhotoData2, data.PhotoData3]; for (let i = 0; i < 3; i++) { if (photoKeys[i] && photoKeys[i].length > 0) { const decoded = Utilities.base64Decode(photoKeys[i]); const blob = Utilities.newBlob(decoded, 'image/jpeg', `Photo_${i+1}_${new Date().getTime()}.jpg`); let folder; const folderIterator = DriveApp.getFoldersByName("SimplyDoors_Vehicle_Photos"); if (folderIterator.hasNext()) { folder = folderIterator.next(); } else { folder = DriveApp.createFolder("SimplyDoors_Vehicle_Photos"); } const file = folder.createFile(blob); photoUrls[i] = file.getUrl(); } } // Format appended photo links for the email const photoText = `\nPhotos Attached:\n1. ${photoUrls[0]}\n2. ${photoUrls[1]}\n3. ${photoUrls[2]}`; // Route Vehicle Inspections if (data.FormType === "Vehicle Inspection") { sheetName = "Vehicle Reports"; headers = ["Timestamp", "Form Type", "Employee", "Vehicle ID", "Odometer", "Critical Issue", "Remarks", "Photo 1", "Photo 2", "Photo 3"]; rowData = [timestamp, data.FormType, data.Employee, data.VehicleID, data.Odometer, data.CriticalIssue, data.Remarks, photoUrls[0], photoUrls[1], photoUrls[2]]; emailSubject = `Vehicle Report: ${data.VehicleID} - ${data.Employee}`; emailBody = `A new vehicle report was submitted.\n\nDriver: ${data.Employee}\nVehicle: ${data.VehicleID}\nCritical Issue: ${data.CriticalIssue}\nRemarks: ${data.Remarks}${photoText}`; } // Route Employee Incident Reports else if (data.FormType === "Employee Incident") { sheetName = "Employee Incident Reports"; headers = ["Timestamp", "Form Type", "Employee", "Job Title", "Department", "Supervisor", "Incident Date", "Incident Time", "Location", "Description", "Witnesses", "Immediate Action", "Root Cause", "Preventative", "Policy", "Corrective Action", "Photo 1", "Photo 2", "Photo 3"]; rowData = [timestamp, data.FormType, data.Employee, data.JobTitle, data.Department, data.Supervisor, data.IncidentDate, data.IncidentTime, data.Location, data.Description, data.Witnesses, data.ImmediateAction, data.RootCause, data.Preventative, data.Policy, data.CorrectiveAction, photoUrls[0], photoUrls[1], photoUrls[2]]; emailSubject = `URGENT: Employee Incident - ${data.Employee}`; emailBody = `An employee incident report was submitted.\n\nEmployee: ${data.Employee}\nDate: ${data.IncidentDate}\nLocation: ${data.Location}\n\nDescription:\n${data.Description}${photoText}`; } // Route Vehicle Incident Reports else if (data.FormType === "Vehicle Incident") { sheetName = "Vehicle Incident Reports"; headers = ["Timestamp", "Form Type", "Employee", "Vehicle ID", "Incident Date", "Incident Time", "Location", "Description", "Police Report", "Other Vehicle", "Photo 1", "Photo 2", "Photo 3"]; rowData = [timestamp, data.FormType, data.Employee, data.VehicleID, data.IncidentDate, data.IncidentTime, data.Location, data.Description, data.PoliceReport, data.OtherVehicle, photoUrls[0], photoUrls[1], photoUrls[2]]; emailSubject = `URGENT: Vehicle Incident - ${data.VehicleID} (${data.Employee})`; emailBody = `A vehicle incident report was submitted.\n\nDriver: ${data.Employee}\nVehicle: ${data.VehicleID}\nDate: ${data.IncidentDate}\nLocation: ${data.Location}\nPolice Report: ${data.PoliceReport}\nOther Vehicle: ${data.OtherVehicle}\n\nDescription:\n${data.Description}${photoText}`; } // Check for existing tab, create if missing, set headers if empty let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold"); } sheet.appendRow(rowData); // Email Routing Logic let recipient = "adem@simplydoors.com, lasanchez67@gmail.com"; if (data.FormType === "Employee Incident" || data.FormType === "Vehicle Incident" || data.CriticalIssue === "YES") { recipient += ", admin@simplydoors.com"; } MailApp.sendEmail(recipient, emailSubject, emailBody); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
