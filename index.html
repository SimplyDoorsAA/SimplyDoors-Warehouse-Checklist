function doPost(e) { try { const data = e.parameter; const timestamp = new Date(); // Create Temporary Google Doc const docName = `SimplyDoors Report - ${data.FormType} - ${data.Employee}`; const doc = DocumentApp.create(docName); const body = doc.getBody(); let emailSubject = ""; // Build Professional Document Header body.appendParagraph(`SIMPLYDOORS OPERATIONS PORTAL (TEST MODE)`).setHeading(DocumentApp.ParagraphHeading.HEADING1); body.appendParagraph(data.FormType.toUpperCase()).setHeading(DocumentApp.ParagraphHeading.HEADING2); body.appendParagraph(`Report Generated: ${timestamp.toLocaleString()}`); body.appendHorizontalRule(); // Inject Data based on Form Type if (data.FormType === "Vehicle Inspection") { emailSubject = `Vehicle Report PDF: ${data.VehicleID} - ${data.Employee}`; body.appendParagraph(`Driver: ${data.Employee}`); body.appendParagraph(`Vehicle: ${data.VehicleID}`); body.appendParagraph(`Odometer: ${data.Odometer}`); body.appendParagraph(`Critical Issue Flagged: ${data.CriticalIssue}`); body.appendParagraph(""); body.appendParagraph("Remarks / Problem Report:").setBold(true); body.appendParagraph(data.Remarks); } else if (data.FormType === "Employee Incident") { emailSubject = `URGENT PDF: Employee Incident - ${data.Employee}`; body.appendParagraph(`Employee: ${data.Employee}`); body.appendParagraph(`Department: ${data.Department}`); body.appendParagraph(`Supervisor: ${data.Supervisor}`); body.appendParagraph(`Date/Time: ${data.IncidentDate} at ${data.IncidentTime}`); body.appendParagraph(`Location: ${data.Location}`); body.appendParagraph(""); body.appendParagraph("Description of Incident:").setBold(true); body.appendParagraph(data.Description); body.appendParagraph(""); body.appendParagraph("Witnesses:").setBold(true); body.appendParagraph(data.Witnesses); body.appendParagraph(""); body.appendParagraph("Immediate Action Taken:").setBold(true); body.appendParagraph(data.ImmediateAction); body.appendParagraph(""); body.appendParagraph("Root Cause Analysis:").setBold(true); body.appendParagraph(data.RootCause); body.appendParagraph(""); body.appendParagraph("Preventative Measures:").setBold(true); body.appendParagraph(data.Preventative); } else if (data.FormType === "Vehicle Incident") { emailSubject = `URGENT PDF: Vehicle Incident - ${data.VehicleID} (${data.Employee})`; body.appendParagraph(`Driver: ${data.Employee}`); body.appendParagraph(`Vehicle Involved: ${data.VehicleID}`); body.appendParagraph(`Date/Time: ${data.IncidentDate} at ${data.IncidentTime}`); body.appendParagraph(`Location: ${data.Location}`); body.appendParagraph(`Police Report Filed: ${data.PoliceReport}`); body.appendParagraph(`Other Vehicle/Property Involved: ${data.OtherVehicle}`); body.appendParagraph(""); body.appendParagraph("Detailed Description of Incident/Damage:").setBold(true); body.appendParagraph(data.Description); } // Process up to 3 Images and append to Document const photoKeys = [data.PhotoData1, data.PhotoData2, data.PhotoData3]; let hasPhotos = false; for (let i = 0; i < 3; i++) { if (photoKeys[i] && photoKeys[i].length > 0) { if (!hasPhotos) { body.appendPageBreak(); body.appendParagraph("Attached Photographs").setHeading(DocumentApp.ParagraphHeading.HEADING3); hasPhotos = true; } const decoded = Utilities.base64Decode(photoKeys[i]); const blob = Utilities.newBlob(decoded, 'image/jpeg', `Photo_${i + 1}.jpg`); const img = body.appendImage(blob); // Scale image proportionally to fit Google Doc margins safely (max ~500px width) const originalWidth = img.getWidth(); if (originalWidth > 500) { const ratio = 500 / originalWidth; img.setWidth(500); img.setHeight(img.getHeight() * ratio); } body.appendParagraph(""); } } // Finalize Document and Convert to PDF doc.saveAndClose(); const pdfBlob = DriveApp.getFileById(doc.getId()).getAs('application/pdf'); pdfBlob.setName(`${data.FormType.replace(" ", "_")}_${data.Employee.replace(" ", "_")}.pdf`); // Core Email Routing Logic - RESTRICTED TO ADEM ONLY FOR TESTING let recipient = "adem@simplydoors.com"; MailApp.sendEmail({ to: recipient, subject: "[TEST] " + emailSubject, body: "This is a test submission. Please see the attached PDF for the full report.", attachments: [pdfBlob] }); // Trash the temporary Google Doc to keep Drive clean DriveApp.getFileById(doc.getId()).setTrashed(true); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
